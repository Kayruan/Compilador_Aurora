programa teste {
	inicio{
		seja n: inteiro;
		seja pares: inteiro = 0;
		seja impares: inteiro = 0;

		para n de 1 ate 100{
			se(n % 2 == 0){
				escreva(n,"e par");
				pares = pares +1;

		}senao{
			escreva(n,"e impar");
			impares = impares +1;
			}
		}
		escreva("\n");
		escreva("total de pares: ", pares);
		escreva("total de impares: ", impares);

	}
}
